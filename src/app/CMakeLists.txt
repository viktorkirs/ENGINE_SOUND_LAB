find_package(Qt6 6.7 REQUIRED COMPONENTS Widgets Concurrent)
qt_standard_project_setup()

add_library(esl_files STATIC preset.cpp audio_exporter.cpp recording_bank.cpp)
target_link_libraries(esl_files PUBLIC esl_dsp Qt6::Core)
set(recording_root "${PROJECT_SOURCE_DIR}/assets/recordings")
file(GLOB_RECURSE recording_files CONFIGURE_DEPENDS "${recording_root}/*.pcm")
qt_add_resources(esl_files field_recordings PREFIX "/recordings" BASE "${recording_root}" FILES ${recording_files} "${recording_root}/banks.json")

qt_add_executable(EngineSoundLab WIN32 main.cpp main_window.cpp preset_library.cpp dashboard.cpp engine_diagram.cpp number_field.cpp export_dialog.cpp audio_output.cpp)
qt_add_resources(esl_files fleet_index PREFIX "/library" BASE "${PROJECT_SOURCE_DIR}/assets" FILES "${PROJECT_SOURCE_DIR}/assets/fleet.json")
file(GLOB preset_files CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/presets/*.json")
qt_add_resources(EngineSoundLab presets PREFIX "/presets" BASE "${PROJECT_SOURCE_DIR}/presets" FILES ${preset_files})
qt_add_resources(EngineSoundLab icons PREFIX "/" FILES icons/chevron-down.png icons/chevron-up.png)
target_link_libraries(EngineSoundLab PRIVATE esl_files Qt6::Widgets Qt6::Concurrent winmm avrt dwmapi)
target_compile_definitions(EngineSoundLab PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN ESL_VERSION="${PROJECT_VERSION}")
set_target_properties(EngineSoundLab PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")

if(BUILD_TESTING)
    add_executable(esl_export_tests "${PROJECT_SOURCE_DIR}/tests/export_tests.cpp")
    target_link_libraries(esl_export_tests PRIVATE esl_files)
    add_test(NAME export COMMAND esl_export_tests "${PROJECT_SOURCE_DIR}/presets")
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(esl_preset_picker_tests "${PROJECT_SOURCE_DIR}/tests/preset_picker_tests.cpp" preset_library.cpp)
    target_link_libraries(esl_preset_picker_tests PRIVATE esl_files Qt6::Widgets Qt6::Test)
    add_test(NAME preset_picker COMMAND esl_preset_picker_tests "${PROJECT_SOURCE_DIR}/presets")
    set_tests_properties(preset_picker PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
endif()
